{{- /* Links for a publication: PDF, DOI, custom links, other url_* fields, and Cite. Context: dict "p" page "cite" bool */ -}} {{- $p := .p -}} {{- $links := slice -}} {{- with $p.Params.url_pdf -}} {{- $u := . -}}{{- if not (findRE `^https?://` $u) }}{{ $u = $u | relURL }}{{ end -}} {{- $links = $links | append (dict "n" "PDF" "u" $u) -}} {{- end -}} {{- with $p.Params.doi }}{{ $links = $links | append (dict "n" "DOI" "u" (printf "https://doi.org/%s" .)) }}{{ end -}} {{- range $p.Params.links -}} {{- if and .name .url -}} {{- $u := .url -}}{{- if not (findRE `^(https?:|mailto:)` $u) }}{{ $u = $u | relURL }}{{ end -}} {{- $links = $links | append (dict "n" .name "u" $u) -}} {{- end -}} {{- end -}} {{- range slice (slice "url_slides" "Slides") (slice "url_video" "Video") (slice "url_code" "Code") (slice "url_dataset" "Data") (slice "url_poster" "Poster") (slice "url_source" "Source") -}} {{- $key := index . 0 -}}{{- $name := index . 1 -}} {{- $u := index $p.Params $key | default "" -}} {{- if $u -}} {{- if not (findRE `^https?://` $u) }}{{ $u = $u | relURL }}{{ end -}} {{- $links = $links | append (dict "n" $name "u" $u) -}} {{- end -}} {{- end -}} {{- if and .cite ($p.Resources.GetMatch "cite.bib") }}{{ $links = $links | append (dict "n" "Cite" "u" (printf "%s#cite" $p.RelPermalink)) }}{{ end -}} {{- $html := slice -}} {{- range $links }}{{ if .n }}{{ $html = $html | append (printf `%s` .u (htmlEscape .n)) }}{{ end }}{{ end -}} {{- delimit $html " · " | safeHTML -}}
